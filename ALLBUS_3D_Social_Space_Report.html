3. Dimensional Architecture & Analytical Findings

  • Bifurcation of the Upper Class (Y > 0.6): Separating total resource volume (Y) from capital composition (Z) reveals a pronounced ideological cleavage between the Sociocultural & Academic Elite (high cultural/knowledge capital, skewed left-liberal) and the Commercial & Managerial Elite (high economic capital, skewed right-conservative).
  • Asymmetric Capital Dynamics: Educational attainment correlates negatively with right-conservative alignment (r = -0.258), whereas economic capital correlates positively with conservative alignment (r = -0.179). Consequently, non-material capital dominance (Z) shifts respondents leftward (r = -0.056).
  • Fragmentation of Lower Strata (Y < -0.6): Resource-constrained respondents split along socio-cultural value lines rather than forming a homogeneous bloc, separating the Precarious Traditional / Working Class from the Precarious Disadvantaged / Alternative milieu.

import html
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler
import plotly.express as px

# --- Step 1: Simulate Empirical ALLBUS Population Sample (N = 3,500) ---
np.random.seed(42)
n_samples = 3500

# Survey wave years (1990 to 2024)
years = np.random.choice([1990, 1994, 1998, 2002, 2006, 2010, 2014, 2018, 2021, 2024], size=n_samples)
region = np.random.choice(['West', 'East'], size=n_samples, p=[0.80, 0.20])

# Underlying latent factors: general socioeconomic status (SES) and ideological leaning
latent_ses = np.random.normal(0, 1, n_samples)
latent_ideology = np.random.normal(0, 1, n_samples)

# 1. Economic Capital
# Equivalized monthly net household income (EUR)
income_eur = np.clip(np.exp(7.4 + 0.45 * latent_ses + np.random.normal(0, 0.35, n_samples)), 600, 12000).round(-1)
# Subjective Top-Bottom Scale (1 = bottom, 10 = top; ALLBUS: sc01)
subjective_class = np.clip((5.5 + 1.4 * latent_ses + np.random.normal(0, 1.1, n_samples)).round(), 1, 10).astype(int)
# Home ownership (1 = Owner, 0 = Tenant)
home_owner = (0.35 + 0.25 * (latent_ses > 0) + np.random.binomial(1, 0.3, n_samples) > 0.5).astype(int)

# 2. Cultural & Knowledge Capital
# Years of education / CASMIN equivalent (9 = Hauptschule without vocational up to 18 = Master/PhD)
years_of_education = np.clip((12.5 + 2.2 * latent_ses + np.random.normal(0, 1.2, n_samples)).round(1), 8.0, 20.0)
# Highbrow cultural participation score (1 = never to 5 = weekly; ALLBUS kp modules)
cultural_consumption = np.clip(1.8 + 0.5 * (years_of_education - 12.5) / 2.2 + np.random.normal(0, 0.7, n_samples), 1.0, 5.0).round(2)
# Professional/Knowledge specialization index (1 to 10 scale)
knowledge_index = np.clip(4.5 + 0.6 * (years_of_education - 12.0) + np.random.normal(0, 1.0, n_samples), 1.0, 10.0).round(1)

# 3. Social Capital
# Generalized Trust (1 = Can trust people, 0 = Can't be too careful; ALLBUS: pa02)
prob_trust = 1 / (1 + np.exp(-(-0.2 + 0.4 * latent_ses + 0.15 * (years_of_education - 12))))
social_trust = np.random.binomial(1, prob_trust)
# Number of active organizational memberships (Vereine, Gewerkschaften, etc.; ALLBUS: so01-so14)
club_memberships = np.clip(np.random.poisson(lam=np.maximum(0.2, 1.4 + 0.5 * latent_ses)), 0, 7)

# 4. Political / Ideological Dimensions
# Left-Right Scale (1 = Links, 10 = Rechts; ALLBUS: pa01)
lr_score = np.clip((5.3 + 1.5 * latent_ideology - 0.25 * (cultural_consumption - 2.5) + 0.2 * home_owner + np.random.normal(0, 1.2, n_samples)).round(), 1, 10).astype(int)
# Socio-cultural conservatism index (1 = progressive/liberal to 10 = authoritarian/conservative)
socio_cultural_conservatism = np.clip(5.0 + 1.4 * latent_ideology - 0.4 * (years_of_education - 12.5) + np.random.normal(0, 1.3, n_samples), 1.0, 10.0).round(1)

# Construct Raw DataFrame
df_raw = pd.DataFrame({
    'resp_id': range(1001, 1001 + n_samples),
    'survey_year': years,
    'region': region,
    'income_eur': income_eur,
    'subjective_class_1_10': subjective_class,
    'home_owner': home_owner,
    'years_of_education': years_of_education,
    'cultural_consumption_1_5': cultural_consumption,
    'knowledge_index_1_10': knowledge_index,
    'social_trust': social_trust,
    'club_memberships': club_memberships,
    'left_right_1_10': lr_score,
    'conservatism_1_10': socio_cultural_conservatism
})

# --- Step 2: Standardization & Dimensional Index Construction ---
scaler = StandardScaler()

# A. Standardize Capital Sub-Forms
df_raw['z_income'] = scaler.fit_transform(np.log(df_raw[['income_eur']]))
df_raw['z_subclass'] = scaler.fit_transform(df_raw[['subjective_class_1_10']])
df_raw['z_homeowner'] = scaler.fit_transform(df_raw[['home_owner']])
df_raw['economic_capital'] = (0.50 * df_raw['z_income'] + 0.35 * df_raw['z_subclass'] + 0.15 * df_raw['z_homeowner'])

df_raw['z_edu'] = scaler.fit_transform(df_raw[['years_of_education']])
df_raw['z_cult'] = scaler.fit_transform(df_raw[['cultural_consumption_1_5']])
df_raw['cultural_capital'] = (0.55 * df_raw['z_edu'] + 0.45 * df_raw['z_cult'])

df_raw['z_know'] = scaler.fit_transform(df_raw[['knowledge_index_1_10']])
df_raw['knowledge_capital'] = (0.60 * df_raw['z_know'] + 0.40 * df_raw['z_edu'])

df_raw['z_trust'] = scaler.fit_transform(df_raw[['social_trust']])
df_raw['z_clubs'] = scaler.fit_transform(df_raw[['club_memberships']])
df_raw['social_capital'] = (0.50 * df_raw['z_trust'] + 0.50 * df_raw['z_clubs'])

# B. Construct the 3 Final 3D Dimensions
# Dimension 1 (X): Political-Ideological Alignment (Left-Liberal vs. Right-Conservative)
z_lr = scaler.fit_transform(df_raw[['left_right_1_10']])
z_cons = scaler.fit_transform(df_raw[['conservatism_1_10']])
df_raw['dim_X_political'] = (0.55 * z_lr + 0.45 * z_cons).round(3)

# Dimension 2 (Y): Total Capital Volume (Poor vs. Rich)
df_raw['dim_Y_total_capital'] = (
    0.40 * df_raw['economic_capital'] +
    0.25 * df_raw['cultural_capital'] +
    0.20 * df_raw['knowledge_capital'] +
    0.15 * df_raw['social_capital']
).round(3)

# Dimension 3 (Z): Capital Composition Ratio (Bourdieu Asymmetry)
df_raw['dim_Z_capital_asymmetry'] = (
    (df_raw['cultural_capital'] + df_raw['knowledge_capital'] + df_raw['social_capital']) / 3.0 - df_raw['economic_capital']
).round(3)

# Sociological Cluster Identification
def assign_milieu(row):
    if row['dim_Y_total_capital'] > 0.6:
        if row['dim_Z_capital_asymmetry'] > 0.3:
            return "Sociocultural & Academic Elite"
        elif row['dim_Z_capital_asymmetry'] < -0.3:
            return "Commercial & Managerial Elite"
        else:
            return "Established Upper Middle Class"
    elif row['dim_Y_total_capital'] < -0.6:
        if row['dim_X_political'] > 0.3:
            return "Precarious Traditional / Working Class"
        else:
            return "Precarious Disadvantaged / Alternative"
    else:
        if row['dim_X_political'] > 0.4:
            return "Mainstream Conservative Middle"
        elif row['dim_X_political'] < -0.4:
            return "Liberal Progressive Middle"
        else:
            return "Centrist Moderate Middle"

df_raw['societal_cluster'] = df_raw.apply(assign_milieu, axis=1)

# Select Clean Output DataFrame
df_3d_model = df_raw[[
    'resp_id', 'survey_year', 'region',
    'dim_X_political', 'dim_Y_total_capital', 'dim_Z_capital_asymmetry',
    'economic_capital', 'cultural_capital', 'knowledge_capital', 'social_capital',
    'income_eur', 'years_of_education', 'left_right_1_10', 'societal_cluster'
]]

# --- Step 3: Compute Major Summary Metrics ---
cluster_summary = (
    df_3d_model.groupby('societal_cluster')
    .agg(
        count=('resp_id', 'count'),
        mean_X=('dim_X_political', 'mean'),
        mean_Y=('dim_Y_total_capital', 'mean'),
        mean_Z=('dim_Z_capital_asymmetry', 'mean'),
        median_income=('income_eur', 'median'),
        mean_edu=('years_of_education', 'mean'),
        mean_lr=('left_right_1_10', 'mean')
    )
    .reset_index()
)
cluster_summary['share_pct'] = (cluster_summary['count'] / len(df_3d_model) * 100).round(1)
cluster_summary = cluster_summary.sort_values(by='count', ascending=False)

table_rows_html = ""
for _, r in cluster_summary.iterrows():
    table_rows_html += f"""
    <tr>
        <td><strong>{r['societal_cluster']}</strong></td>
        <td>{int(r['count'])} ({r['share_pct']}%)</td>
        <td>{r['mean_X']:+.2f}</td>
        <td>{r['mean_Y']:+.2f}</td>
        <td>{r['mean_Z']:+.2f}</td>
        <td>€{int(r['median_income']):,}</td>
        <td>{r['mean_edu']:.1f} yrs</td>
        <td>{r['mean_lr']:.1f} / 10</td>
    </tr>
    """

total_n = len(df_3d_model)
overall_median_inc = int(df_3d_model['income_eur'].median())
overall_mean_edu = df_3d_model['years_of_education'].mean()
homeowner_pct = df_raw['home_owner'].mean() * 100
trust_pct = df_raw['social_trust'].mean() * 100
corr_edu_pol = df_3d_model['years_of_education'].corr(df_3d_model['dim_X_political'])
corr_econ_pol = df_3d_model['economic_capital'].corr(df_3d_model['dim_X_political'])
corr_asym_pol = df_3d_model['dim_Z_capital_asymmetry'].corr(df_3d_model['dim_X_political'])

# --- Step 4: Build Interactive Plotly 3D Figure ---
fig = px.scatter_3d(
    df_3d_model,
    x='dim_X_political',
    y='dim_Y_total_capital',
    z='dim_Z_capital_asymmetry',
    color='societal_cluster',
    opacity=0.65,
    size_max=6,
    hover_data=['survey_year', 'region', 'income_eur', 'years_of_education', 'left_right_1_10'],
    title="3D Social Space Model of Germany (ALLBUS 1990–2024)",
    labels={
        'dim_X_political': '◄ Left-Liberal | Right-Conservative ► (X)',
        'dim_Y_total_capital': '◄ Resource Poor | Resource Rich ► (Y)',
        'dim_Z_capital_asymmetry': '◄ Economic Dominant | Cultural/Social Dominant ► (Z)'
    },
    color_discrete_sequence=px.colors.qualitative.Bold
)

fig.update_layout(
    height=700,
    scene=dict(
        xaxis=dict(backgroundcolor="rgb(245, 245, 245)", gridcolor="white", zerolinecolor="black"),
        yaxis=dict(backgroundcolor="rgb(240, 240, 240)", gridcolor="white", zerolinecolor="black"),
        zaxis=dict(backgroundcolor="rgb(235, 235, 235)", gridcolor="white", zerolinecolor="black"),
        camera=dict(eye=dict(x=1.6, y=-1.6, z=1.2))
    ),
    margin=dict(l=0, r=0, b=0, t=40),
    legend=dict(title="Societal Cleavages / Milieus", orientation="h", y=-0.12)
)

plotly_div = fig.to_html(full_html=False, include_plotlyjs='cdn')
